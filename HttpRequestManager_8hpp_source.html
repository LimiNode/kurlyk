47 SubmitResult submit_request(
48 std::unique_ptr<HttpRequest> request_ptr,
49 HttpResponseCallback callback) {
50 std::lock_guard<std::mutex> lock(m_mutex);
51 if (m_shutdown) {
52 return SubmitResult{false, utils::make_error_code(utils::ClientError::ShuttingDown)};
53 }
54
55 const std::size_t queue_limit = m_max_pending_requests.load();
56 if (queue_limit && m_pending_requests.size() >= queue_limit) {
57 return SubmitResult{false, utils::make_error_code(utils::ClientError::QueueLimitExceeded)};
58 }
59
60# if __cplusplus >= 201402L
61 auto context = std::make_unique<HttpRequestContext>(std::move(request_ptr), std::move(callback));
62# else
63 auto context = std::unique_ptr<HttpRequestContext>(
64 new HttpRequestContext(std::move(request_ptr), std::move(callback)));
65# endif
66 context->in_flight_token = m_next_in_flight_token.fetch_add(1, std::memory_order_relaxed);
67 const uint64_t group_id = context->request ? context->request->group_id : 0;
68 if (group_id != 0) {
69 context->on_group_complete = [this, group_id]() {
70 complete_group_request(group_id);
71 };
72 ++m_group_request_counts[group_id];
73 }
74 try {
75 m_pending_requests.push_back(std::move(context));
76 } catch (...) {
77 if (group_id != 0) {
78 auto count_it = m_group_request_counts.find(group_id);
79 if (count_it != m_group_request_counts.end()) {
80 if (count_it->second > 1) {
81 --count_it->second;
82 } else {
83 m_group_request_counts.erase(count_it);
84 }
85 }
86 }

88 }

427 void process_pending_requests() {
428 std::unique_lock<std::mutex> lock(m_mutex);
429 if (m_pending_requests.empty()) return;
430
431 std::vector<std::unique_ptr<HttpRequestContext>> pending_request;
432 std::vector<std::unique_ptr<HttpRequestContext>> failed_requests;
433
434 auto it = m_pending_requests.begin();
435 while (it != m_pending_requests.end()) {
436 auto& context = *it;
437 auto& request = context->request;
438 // Check if the request is valid.
439 if (!request) {
440 failed_requests.push_back(std::move(context));
441 it = m_pending_requests.erase(it);
442 continue;
443 }
444
445 // Set up completion callback before allow_request so it is armed
446 // even if an exception occurs after the token is committed.
447 auto general_limit = request->general_rate_limit;
448 auto specific_limit = request->specific_rate_limit;
449 uint64_t token = context->in_flight_token;
450 auto general_key = request->general_rate_limit_key;
451 auto specific_key = request->specific_rate_limit_key;
452
453 // Preserve any previous on_complete so a retry that already owns
454 // a sequential token does not lose its cleanup callback.
455 auto old_on_complete = std::move(context->on_complete);
456 context->on_complete = [this, general_limit, specific_limit, token, general_key, specific_key]() {
457 m_rate_limiter.release_request(general_limit, specific_limit, token, general_key, specific_key);
458 };
459
460 // Check if the request is allowed by the rate limiter.
461 const bool allowed = m_rate_limiter.allow_request(
462 general_limit,
463 specific_limit,
464 token,
465 request->general_rate_limit_key,
466 request->specific_rate_limit_key);
467 if (!allowed) {
468 context->on_complete = std::move(old_on_complete);
469 ++it;
470 continue;
471 }
472
473 pending_request.push_back(std::move(context));
474 it = m_pending_requests.erase(it);
475 }
476 lock.unlock();
477
478 // Handle failed requests by calling their callback with a 400 status.
479 if (!failed_requests.empty()) {
480 for (const auto &context : failed_requests) {
481# if __cplusplus >= 201402L
482 auto response = std::make_unique<HttpResponse>();
483# else
484 auto response = std::unique_ptr<HttpResponse>(new HttpResponse());
485# endif
486 const long BAD_REQUEST = 400;
487 response->error_code = utils::make_error_code(CURLE_OK);
488 response->status_code = BAD_REQUEST;
489 response->ready = true;
490 context->callback(std::move(response));
491 context->complete();
492 }
493 failed_requests.clear();
494 }
495
496 // If there are ready requests, create a new HttpBatchRequestHandler to manage them.
497 if (pending_request.empty()) return;
498 try {
499# if __cplusplus >= 201402L
500 m_active_request_batches.push_back(std::make_unique<HttpBatchRequestHandler>(pending_request));
501# else
502 m_active_request_batches.push_back(std::unique_ptr<HttpBatchRequestHandler>(new HttpBatchRequestHandler(pending_request)));
503# endif
504 } catch (...) {
505 for (auto& context : pending_request) {
506 if (!context || !context->callback) continue;
507# if __cplusplus >= 201402L
508 auto response = std::make_unique<HttpResponse>();
509# else
510 auto response = std::unique_ptr<HttpResponse>(new HttpResponse());
511# endif
512 response->error_code = utils::make_error_code(utils::ClientError::AbortedDuringDestruction);
513 response->status_code = 499; // Client closed request
514 response->ready = true;
515 context->callback(std::move(response));
516 context->complete();
517 }
518 return;
519 }
520 }

540 void process_retry_failed_requests() {
541 auto it = m_failed_requests.begin();
542 while (it != m_failed_requests.end()) {
543 auto& request_context = *it;
544 if (!request_context || !request_context->request) {
545 it = m_failed_requests.erase(it);
546 continue;
547 }
548
549 const auto now = std::chrono::steady_clock::now();
550 const auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(now - request_context->start_time);
551 const auto& retry_delay_ms = request_context->request->retry_delay_ms;
552 if (duration.count() >= retry_delay_ms) {
553 std::unique_lock<std::mutex> lock(m_mutex);
554 m_pending_requests.push_back(std::move(request_context));
555 lock.unlock();
556 it = m_failed_requests.erase(it);
557 continue;
558 }
559 ++it;
560 }

564 void process_cancel_requests() {
565 std::unique_lock<std::mutex> lock(m_mutex);
566 if (m_requests_to_cancel_by_id.empty() && m_groups_to_cancel.empty()) return;
567
568 auto requests_to_cancel = std::move(m_requests_to_cancel_by_id);
569 auto groups_to_cancel = std::move(m_groups_to_cancel);
570 m_requests_to_cancel_by_id.clear();
571 m_groups_to_cancel.clear();
572
573 std::list<std::unique_ptr<HttpRequestContext>> canceled_pending_requests;
574 auto pending_it = m_pending_requests.begin();
575 while (pending_it != m_pending_requests.end()) {
576 const auto& ctx = *pending_it;
577 if (!matches_cancel(ctx, requests_to_cancel, groups_to_cancel)) {
578 ++pending_it;
579 continue;
580 }
581 canceled_pending_requests.push_back(std::move(*pending_it));
582 pending_it = m_pending_requests.erase(pending_it);
583 }
584 lock.unlock();
585
586 for (const auto& request_context : canceled_pending_requests) {
587 request_context->callback(make_cancelled_response());
588 request_context->complete();
589 }
590
591 auto failed_it = m_failed_requests.begin();
592 while (failed_it != m_failed_requests.end()) {
593 const auto& request_context = *failed_it;
594 if (!matches_cancel(request_context, requests_to_cancel, groups_to_cancel)) {
595 ++failed_it;
596 continue;
597 }
598 request_context->callback(make_cancelled_response());
599 request_context->complete();
600 failed_it = m_failed_requests.erase(failed_it);
601 }
602
603 for (const auto &handler : m_active_request_batches) {
604 handler->cancel_requests(requests_to_cancel, groups_to_cancel);
605 }
606
607 invoke_cancel_callbacks(requests_to_cancel);
608 invoke_cancel_callbacks(groups_to_cancel);
609 }

611 static bool matches_cancel(
612 const std::unique_ptr<HttpRequestContext>& ctx,
613 const cancel_map_t& requests_to_cancel,
614 const cancel_map_t& groups_to_cancel) {
615 if (!ctx || !ctx->request) return false;
616 const auto request_id = ctx->request->request_id;
617 const auto group_id = ctx->request->group_id;
618 return
619 (request_id != 0 && requests_to_cancel.count(request_id) > 0) ||
620 (group_id != 0 && groups_to_cancel.count(group_id) > 0);
621 }
