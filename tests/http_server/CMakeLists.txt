cmake_minimum_required(VERSION 3.21)
project(kurlyk_http_server_tests LANGUAGES CXX)
enable_testing()
set(KURLYK_SIMPLE_WEB_SERVER_SOURCE_DIR "" CACHE PATH "Pinned Simple-Web-Server source")
set(KURLYK_ASIO_SOURCE_DIR "" CACHE PATH "Pinned standalone Asio source")
if(NOT KURLYK_SIMPLE_WEB_SERVER_SOURCE_DIR)
    set(KURLYK_SIMPLE_WEB_SERVER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../external/Simple-Web-Server")
endif()
if(NOT KURLYK_ASIO_SOURCE_DIR)
    set(KURLYK_ASIO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../external/asio")
endif()
if(NOT TARGET kurlyk::http_server)
    if(NOT EXISTS "${KURLYK_SIMPLE_WEB_SERVER_SOURCE_DIR}/server_http.hpp" OR
       NOT EXISTS "${KURLYK_ASIO_SOURCE_DIR}/include/asio.hpp")
        message(FATAL_ERROR "Set KURLYK_SIMPLE_WEB_SERVER_SOURCE_DIR and KURLYK_ASIO_SOURCE_DIR to pinned local sources")
    endif()
    add_library(kurlyk_http_server_test_headers INTERFACE)
    target_include_directories(kurlyk_http_server_test_headers INTERFACE
        ../../include "${KURLYK_SIMPLE_WEB_SERVER_SOURCE_DIR}" "${KURLYK_ASIO_SOURCE_DIR}/include")
    target_compile_definitions(kurlyk_http_server_test_headers INTERFACE ASIO_STANDALONE)
    add_library(kurlyk::http_server ALIAS kurlyk_http_server_test_headers)
endif()
find_package(Threads REQUIRED)
add_executable(http_server_test main.cpp)
target_link_libraries(http_server_test PRIVATE kurlyk::http_server Threads::Threads)
if(WIN32)
    target_link_libraries(http_server_test PRIVATE ws2_32 mswsock)
endif()
add_test(NAME http_server_foundation COMMAND http_server_test)
set_tests_properties(http_server_foundation PROPERTIES TIMEOUT 30)
