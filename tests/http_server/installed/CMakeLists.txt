cmake_minimum_required(VERSION 3.21)
project(kurlyk_installed_http_server LANGUAGES CXX)
enable_testing()
find_package(kurlyk CONFIG REQUIRED COMPONENTS http_server)
add_executable(installed_http_server ../main.cpp)
target_link_libraries(installed_http_server PRIVATE kurlyk::http_server)
add_test(NAME installed_http_server COMMAND installed_http_server)
set_tests_properties(installed_http_server PROPERTIES TIMEOUT 30)
add_executable(combined_client_server combined.cpp)
target_link_libraries(combined_client_server PRIVATE kurlyk::kurlyk kurlyk::http_server)
add_test(NAME combined_client_server COMMAND combined_client_server)
set_tests_properties(combined_client_server PROPERTIES TIMEOUT 5)
