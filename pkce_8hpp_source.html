69 inline std::string generate_code_verifier(std::size_t length = 64) {
70 static const char allowed[] =
71 "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_.~";
72 if (length < 43) length = 43;
73 if (length > 128) length = 128;
74
75 std::string verifier;
76 verifier.reserve(length);
77
78 const std::size_t allowed_count = sizeof(allowed) - 1; // exclude null terminator
79 const std::size_t accepted_limit = 256 - (256 % allowed_count);
80
81 while (verifier.size() < length) {
82 const std::size_t remaining = length - verifier.size();
83 std::vector<uint8_t> random_data(remaining);
84 detail::random_bytes(random_data.data(), random_data.size());
85
86 for (std::size_t i = 0; i < random_data.size() && verifier.size() < length; ++i) {
87 if (random_data[i] >= accepted_limit) {
88 continue;
89 }
90 verifier.push_back(allowed[random_data[i] % allowed_count]);
91 }
92 }
93 return verifier;
