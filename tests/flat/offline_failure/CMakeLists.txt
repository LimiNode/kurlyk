cmake_minimum_required(VERSION 3.21)
project(kurlyk_offline_failure LANGUAGES CXX)
set(KURLYK_ALLOW_DOWNLOADS OFF)
add_library(probe INTERFACE)
if(NOT KURLYK_TEST_LOADER)
    message(FATAL_ERROR "KURLYK_TEST_LOADER is required")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/deps/fallbacks/${KURLYK_TEST_LOADER}.cmake")
if(KURLYK_TEST_LOADER STREQUAL "load_asio")
    load_asio(probe)
elseif(KURLYK_TEST_LOADER STREQUAL "load_simple_ws_server")
    load_simple_ws_server(probe)
elseif(KURLYK_TEST_LOADER STREQUAL "load_simple_web_server")
    load_simple_web_server(probe)
elseif(KURLYK_TEST_LOADER MATCHES "load_curl_")
    load_curl(probe)
else()
    load_openssl(probe)
endif()
