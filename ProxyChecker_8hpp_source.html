22 inline bool proxy_check_has_scheme(const std::string& url, const char* scheme) {
23 const std::string prefix(scheme);
24 if (url.size() < prefix.size()) return false;
25 for (std::string::size_type i = 0; i < prefix.size(); ++i) {
26 char actual = url[i];
27 if (actual >= 'A' && actual <= 'Z') {
28 actual = static_cast<char>(actual - 'A' + 'a');
29 }
30 if (actual != prefix[i]) return false;
31 }
32 return true;
33 }

40 inline long proxy_check_milliseconds(double seconds) {
41 if (seconds <= 0.0) return 0;
42 const double milliseconds = seconds * 1000.0;
43 const double maximum = static_cast<double>((std::numeric_limits<long>::max)());
44 if (milliseconds >= maximum) {
45 return (std::numeric_limits<long>::max)();
46 }
47 return static_cast<long>(milliseconds + 0.5);

59 inline ProxyCheckResult make_proxy_check_result(
60 const HttpResponsePtr& response,
61 bool is_https) {
62 if (!response) {
63 return make_proxy_check_error(
64 utils::make_error_code(utils::ClientError::AbortedDuringDestruction),
65 "Proxy check returned no HTTP response");
66 }
67
68 ProxyCheckResult result;
69 result.error_code = response->error_code;
70 result.error_message = response->error_message.empty() && response->error_code
71 ? response->error_code.message()
72 : response->error_message;
73 result.http_status = response->status_code;
74
75 const bool received_http_response =
76 response->ready &&
77 response->status_code > 0 &&
78 (!response->error_code || utils::is_http_error(response->error_code));
79 result.reachable = received_http_response;
80 result.http_ok = received_http_response &&
81 response->status_code >= 200 && response->status_code < 400;
82 result.https_ok = is_https && result.http_ok;
83
84 result.connect_latency_ms = proxy_check_milliseconds(response->connect_time);
85 if (response->appconnect_time > 0.0 &&
86 response->connect_time >= 0.0 &&
87 response->appconnect_time >= response->connect_time) {
88 result.tls_latency_ms = proxy_check_milliseconds(
89 response->appconnect_time - response->connect_time);
90 }
91 result.ttfb_ms = proxy_check_milliseconds(response->starttransfer_time);
92 result.total_latency_ms = proxy_check_milliseconds(response->total_time);
93 return result;
94 }
