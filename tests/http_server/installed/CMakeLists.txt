cmake_minimum_required(VERSION 3.21)
project(kurlyk_installed_http_server LANGUAGES CXX)
enable_testing()
set(KURLYK_TEST_SIMPLE_WEB_SERVER_SOURCE_DIR "" CACHE PATH "Flat parent-provided HTTP server headers")
if(KURLYK_TEST_SIMPLE_WEB_SERVER_SOURCE_DIR AND NOT TARGET simple_web_server)
    add_library(simple_web_server INTERFACE)
    target_include_directories(simple_web_server INTERFACE "${KURLYK_TEST_SIMPLE_WEB_SERVER_SOURCE_DIR}")
endif()
find_package(kurlyk CONFIG REQUIRED COMPONENTS http_server)
get_target_property(kurlyk_installed_include_dirs kurlyk::http_server INTERFACE_INCLUDE_DIRECTORIES)
add_executable(installed_http_server_data ../data_smoke.cpp)
target_include_directories(installed_http_server_data PRIVATE ${kurlyk_installed_include_dirs})
add_test(NAME installed_http_server_data_standalone COMMAND installed_http_server_data)
add_executable(installed_http_server_contract ../contract.cpp)
target_include_directories(installed_http_server_contract PRIVATE ${kurlyk_installed_include_dirs})
add_test(NAME installed_http_server_backend_contract COMMAND installed_http_server_contract)
set_tests_properties(installed_http_server_data_standalone installed_http_server_backend_contract PROPERTIES TIMEOUT 5)
add_executable(installed_http_server ../main.cpp)
# Avoid Windows installer detection for test executables; target/test names stay stable.
set_target_properties(installed_http_server PROPERTIES OUTPUT_NAME http_server_consumer)
set_target_properties(installed_http_server_data PROPERTIES OUTPUT_NAME http_server_data_consumer)
set_target_properties(installed_http_server_contract PROPERTIES OUTPUT_NAME http_server_contract_consumer)
target_link_libraries(installed_http_server PRIVATE kurlyk::http_server)
add_test(NAME installed_http_server COMMAND installed_http_server)
set_tests_properties(installed_http_server PROPERTIES TIMEOUT 30)
add_executable(combined_client_server combined.cpp)
target_link_libraries(combined_client_server PRIVATE kurlyk::kurlyk kurlyk::http_server)
add_test(NAME combined_client_server COMMAND combined_client_server)
set_tests_properties(combined_client_server PROPERTIES TIMEOUT 5)
