52 void cancel_requests() {
53 auto& worker = core::NetworkWorker::get_instance();
54 auto& manager = HttpRequestManager::get_instance();
55 const uint64_t group_id = m_request.group_id;
56 if (worker.is_worker_thread()) {
57 // On the worker thread: execute cancellation inline to avoid self-deadlock.
58 manager.cancel_requests_by_group_id(group_id, nullptr);
59 worker.process();
60 return;
61 }
62 auto promise = std::make_shared<std::promise<void>>();
63 auto future = promise->get_future();
64 manager.cancel_requests_by_group_id(group_id, [promise](){
65 try {
66 promise->set_value();
67 } catch (const std::future_error& e) {
68 if (e.code() == std::make_error_condition(std::future_errc::promise_already_satisfied)) {
69 KURLYK_HANDLE_ERROR(e, "Promise already satisfied in HttpClient::request callback");
70 } else {
71 KURLYK_HANDLE_ERROR(e, "Future error in HttpClient::request callback");
72 }
73 } catch (const std::exception& e) {
74 KURLYK_HANDLE_ERROR(e, "Unhandled exception in HttpClient::request callback");
75 } catch (...) {
76 // Unknown fatal error in request callback
77 }
78 });
79 worker.notify();
80 try {
81 future.get();
82 } catch (const std::exception& e) {
83 KURLYK_HANDLE_ERROR(e, "cancel_requests() future.get() failed");
84 }

109 bool wait_requests_for(std::chrono::milliseconds timeout) {
110 auto& worker = core::NetworkWorker::get_instance();
111 if (worker.is_worker_thread()) {
112 return false;
113 }
114 uint64_t waiter_id = 0;
115 auto future = make_wait_requests_future(&waiter_id);
116 worker.notify();
117 if (future.wait_for(timeout) == std::future_status::timeout) {
118 HttpRequestManager::get_instance().cancel_wait_requests_by_group_id(
119 m_request.group_id,
120 waiter_id);
121 return false;
122 }
123 try {
124 future.get();
125 } catch (const std::exception& e) {
126 KURLYK_HANDLE_ERROR(e, "wait_requests_for() future.get() failed");
127 return false;
128 }

587 SubmitResult submit_request(
588 std::unique_ptr<HttpRequest> request_ptr,
589 HttpResponseCallback callback) {
590 SubmitResult submit_result;
591 {
592 std::lock_guard<std::mutex> lock(m_submit_mutex);
593
594 if (m_max_in_flight != 0 &&
595 HttpRequestManager::get_instance().group_request_count(m_request.group_id) >= m_max_in_flight) {
596 return SubmitResult{false, utils::make_error_code(utils::ClientError::QueueLimitExceeded)};
597 }
598
599 submit_result = HttpRequestManager::get_instance().submit_request(
600 std::move(request_ptr), std::move(callback));
601 }
602
603 if (submit_result) {
604 core::NetworkWorker::get_instance().notify();
605 }
606 return submit_result;
607 }

1062 void safe_submit_request(
1063 std::shared_ptr<std::promise<HttpResponsePtr>> promise,
1064 std::unique_ptr<HttpRequest> request_ptr,
1065 HttpResponseCallback callback) {
1066 try {
1067 const SubmitResult submit_result = submit_request(std::move(request_ptr), std::move(callback));
1068 if (!submit_result) {
1069 safe_set_response(promise, make_submit_error_response(submit_result));
1070 }
1071 } catch (const std::exception& e) {
1072 KURLYK_HANDLE_ERROR(e, "Exception while submitting request");
1073 try {
1074 promise->set_exception(std::current_exception());
1075 } catch (...) {} // fallback
1076 } catch (...) {
